# Cybernetic Creed leader-trait sync + Cyberization situation deadlock recovery.
#
# Source: ~~Stellaris [v4.3] General Fixes
#         - events/cyber_creed_start_fix.txt (machine_age.2151/2152, by FirePrince)
#         - events/cyber_machine_age_situation_events_fix.txt (cyber_patch.18/19, by FirePrince)
#
# Both fixes are no-ops on healthy saves; only fire when the broken state
# (trait mismatch / stuck augmentation_center_needed flag) is detected.
#
# Requires Machine Age DLC content (origin_cybernetic_creed, situation_cyberization,
# building_augmentation_center). Triggers will simply not match without the DLC.

namespace = smf

# === smf.30: Cybernetic Creed leader trait sync ===
# Dispatcher — fires per save load for every Cybernetic Creed empire
event = {
	id = smf.30
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		every_playable_country = {
			limit = { has_origin = origin_cybernetic_creed }
			country_event = { id = smf.301 days = 3 }
		}
	}
}

country_event = {
	id = smf.301
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_cybernetic_creed
	}

	immediate = {
		# Remove ritualistic-implants trait from non-spiritualist non-rulers
		every_owned_leader = {
			limit = {
				NOR = {
					is_spiritualist = yes
					is_ruler = yes
				}
				has_trait = leader_trait_ritualistic_implants_cyborg
			}
			remove_trait = leader_trait_ritualistic_implants_cyborg
			if = {
				limit = { species = { has_trait = trait_cyborg_ritualistic_implants } }
				prev = {
					random_owned_species = {
						limit = {
							is_same_species = prevprev.species
							NOT = { has_trait = trait_cyborg_ritualistic_implants }
						}
						weights = {
							base = 1
							modifier = { add = 1000 has_cybernetic_creed_trait = yes }
						}
						prevprev = { change_species = prev update_leader_after_modification = yes }
					}
				}
			}
		}
		every_pool_leader = {
			limit = {
				is_spiritualist = no
				has_trait = leader_trait_ritualistic_implants_cyborg
			}
			remove_trait = leader_trait_ritualistic_implants_cyborg
			if = {
				limit = { species = { has_trait = trait_cyborg_ritualistic_implants } }
				prev = {
					random_owned_species = {
						limit = {
							is_same_species = prevprev.species
							NOT = { has_trait = trait_cyborg_ritualistic_implants }
						}
						weights = {
							base = 1
							modifier = { add = 1000 has_cybernetic_creed_trait = yes }
						}
						prevprev = { change_species = prev update_leader_after_modification = yes }
					}
				}
			}
		}
		# Add ritualistic-implants trait to spiritualists / rulers without it
		every_owned_leader = {
			limit = {
				OR = {
					is_spiritualist = yes
					is_ruler = yes
				}
				NOR = {
					has_trait = leader_trait_ritualistic_implants_cyborg
					has_trait = leader_trait_psionic
					has_trait = leader_trait_chosen
					has_trait = leader_trait_cyborg
				}
			}
			add_trait = {
				trait = leader_trait_ritualistic_implants_cyborg
				show_message = no
			}
			if = {
				limit = { species = { has_cybernetic_creed_trait = no } }
				prev = {
					random_owned_species = {
						limit = {
							is_same_species = prevprev.species
							has_cybernetic_creed_trait = yes
						}
						weights = {
							base = 1
							modifier = {
								add = 1000
								has_trait = leader_trait_ritualistic_implants_cyborg
							}
						}
						prevprev = { change_species = prev update_leader_after_modification = yes }
					}
				}
			}
		}
		every_pool_leader = {
			limit = {
				is_spiritualist = yes
				NOR = {
					has_trait = leader_trait_ritualistic_implants_cyborg
					has_trait = leader_trait_psionic
					has_trait = leader_trait_chosen
					has_trait = leader_trait_cyborg
				}
			}
			add_trait = {
				trait = leader_trait_ritualistic_implants_cyborg
				show_message = no
			}
		}
	}
}

# === smf.31: Cyberization situation deadlock recovery ===
# Vanilla bug: a stale `augmentation_center_needed` flag prevents the
# Cyberization situation from completing even after the building is up.
# Detector: any country with both `augmentation_center_available` AND
# `augmentation_center_needed` flags set, where the situation is already
# complete or the tradition is already fully unlocked.
event = {
	id = smf.31
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_playable_country = {
			has_tradition = tr_cybernetics_adopt
			has_country_flag = augmentation_center_available
			has_country_flag = augmentation_center_needed
			OR = {
				has_country_flag = situation_cyberization_complete
				has_country_flag = cybernetics_traditions_unlocked
				AND = {
					is_hive_empire = no
					country_has_situation = { SITUATION = situation_cyberization }
				}
				AND = {
					is_hive_empire = yes
					country_has_situation = { SITUATION = situation_cyberization_hive }
				}
			}
		}
	}

	immediate = {
		random_playable_country = {
			limit = {
				has_tradition = tr_cybernetics_adopt
				has_country_flag = augmentation_center_available
				has_country_flag = augmentation_center_needed
				OR = {
					has_country_flag = situation_cyberization_complete
					has_country_flag = cybernetics_traditions_unlocked
					AND = {
						is_hive_empire = no
						country_has_situation = { SITUATION = situation_cyberization }
					}
					AND = {
						is_hive_empire = yes
						country_has_situation = { SITUATION = situation_cyberization_hive }
					}
				}
			}
			weights = {
				base = 1
				modifier = { add = 99 is_ai = no }
				modifier = {
					add = 99
					any_owned_planet = {
						is_colony = yes
						is_occupied_flag = no
						has_building = yes
						has_active_building = building_augmentation_center
					}
				}
			}
			country_event = { id = smf.311 days = 3 }
		}
	}
}

# Clears the stuck flag if the building is actually present
country_event = {
	id = smf.311
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = augmentation_center_needed
		any_owned_planet = {
			is_colony = yes
			is_occupied_flag = no
			has_building = yes
			has_active_building = building_augmentation_center
		}
	}

	immediate = {
		remove_country_flag = augmentation_center_needed
		log = "SMF: cleared stuck augmentation_center_needed flag on [This.GetName]"
	}
}
